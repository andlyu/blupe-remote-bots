"""BluPe-only deployment wiring; shared experiment source remains in codex-runner."""
import asyncio
import importlib.util
import json
import os
from pathlib import Path
import sys
import threading
from urllib.parse import urlsplit

from guest_runs import GuestRuns, GuestLimitReached


def hosted_runner_type(experiment_type, allowance):
    from playground import RequestError

    class HostedBrightness(experiment_type):
        def __init__(self, **kwargs):
            self._allowance_lock = threading.RLock()
            super().__init__(**kwargs)

        def remaining(self, visitor):
            return dict(limit=allowance.limit, remaining=allowance.remaining(
                getattr(visitor, 'guest_token', '')))

        async def session_extension(self, cookies, visitor):
            setup, headers = await super().session_extension(cookies, visitor)
            saved = cookies.get('__Host-blupe-guest')
            visitor.guest_token = await asyncio.to_thread(allowance.visitor,
                getattr(visitor, 'guest_token', '') or (saved.value if saved else ''))
            cookie = ('__Host-blupe-guest=' + visitor.guest_token +
                '; Path=/; HttpOnly; Secure; SameSite=Lax; Max-Age=31536000')
            return {**setup, 'experiment_allowance':self.remaining(visitor)}, [
                *headers, (b'set-cookie', cookie.encode())]

        def launch(self, visitor, payload, **kwargs):
            with self._allowance_lock:
                try:
                    result = allowance.launch(getattr(visitor, 'guest_token', ''),
                        lambda:super(HostedBrightness, self).launch(visitor, payload, **kwargs))
                except GuestLimitReached:
                    raise RequestError(429, 'This browser has used its experiment guest runs.') from None
                return {**result, 'experiment_allowance':self.remaining(visitor)}

    return HostedBrightness


def create_app():
    if os.environ.get('YAM_BRIGHTNESS_IDENTITY_FILE'):
        raise ValueError('Private runner defaults must not be configured on a public service')
    origin = os.environ['YAM_BRIGHTNESS_ORIGIN']
    parsed = urlsplit(origin)
    if (parsed.scheme != 'https' or not parsed.hostname or parsed.path not in ('', '/')
            or parsed.username or parsed.password or parsed.query or parsed.fragment):
        raise ValueError('The hosted experiment requires one explicit HTTPS origin')
    shared = Path(os.environ['BLUPE_REMOTE_YAM_ROOT']).resolve()
    extension = shared / 'integrations/dream-brightness'
    sys.path[:0] = [str(extension), str(shared), str(shared / 'src')]
    spec = importlib.util.spec_from_file_location('dream_brightness_extension', extension / 'serve.py')
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    state = Path(os.environ['YAM_BRIGHTNESS_STATE']).resolve()
    state.mkdir(parents=True, exist_ok=True, mode=0o700)
    for variable, filename in [('YAM_SOCIAL_DATABASE','social.sqlite3'),
        ('YAM_RUN_NAMES_DATABASE','names.sqlite3'), ('YAM_SUBMISSION_DATABASE','submissions.sqlite3'),
        ('YAM_ASSESSMENT_DATABASE','assessments.sqlite3'), ('YAM_TASK_CLOUD_DATABASE','task-cloud.sqlite3')]:
        os.environ[variable] = str(state / filename)
    cache = state / 'jax'; cache.mkdir(exist_ok=True, mode=0o700)
    os.environ['JAX_COMPILATION_CACHE_DIR'] = str(cache)
    allowance = GuestRuns(state / 'guests.sqlite3', limit=10)
    runner = hosted_runner_type(module.make_runner_type(shared, local=False), allowance)
    robots = [robot for robot in json.loads((shared / 'config/robots.json').read_text())
        if robot['id'] == 'robot-ba8413962083809c']
    from multi_robot import fleet
    return fleet(runner, dict(public_origin=origin.rstrip('/'),
        session_api=module.BACKEND, camera_origin=module.BACKEND,
        hardware_control=True, development=False, max_sessions=32,
        default_provider='dream', dream_credentials_file=os.environ['YAM_DREAM_CREDENTIALS'],
        dream_python=os.environ['YAM_DREAM_PYTHON'],
        dream_worker_pool=threading.BoundedSemaphore(1),
        chat_database=str(state / 'chat.sqlite3'), experiment_root=state / 'experiment'), robots)
